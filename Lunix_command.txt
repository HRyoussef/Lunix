>> a partire du pdf (concat.pdf) genère un nuveau pdf qui contien les 10 premier pages du pdf original

>> extraire les images dans un pdf (concat.pdf) en format .jpg, .ppm, .pbm
 pdfimages -j concat.pdf prefix

>> commande (script shell) pour converter les ppm et pbm en jpg
for f in *.{ppm,pbm}; do     [ -f "$f" ] || continue      # ignore si le motif ne trouve rien
    filename=$(basename "$f")    # ne prend que le nom du fichier
    convert "$f" "pdf_image_jpg/${filename%.*}.jpg"; done
